#include <iostream>
using namespace std;

class Shape {
public:
    virtual void area() {
        cout << "Calculating area of shape" << endl;
    }
};

class Circle : public Shape {
public:
    void area() {
        cout << "Area of Circle = pi * r * r" << endl;
    }
};

class Rectangle : public Shape {
public:
    void area() {
        cout << "Area of Rectangle = length * width" << endl;
    }
};

int main() {
    Shape* s;
    Circle c;
    Rectangle r;

    s = &c;
    s->area();

    s = &r;
    s->area();

    return 0;
}
